/**
 * Question Registry — Intelligent Question Bank
 * 
 * Three-layer system:
 * 1. Universal: default included, blocked by CE signals
 * 2. Category-specific: gated by subcategory + trigger signals
 * 3. CE-specific: gated by DRD signals
 * 
 * Each question generates: fallback text (SEO) + embed spec (iframe) + editorial overlay (CTA)
 */

import { z } from "zod";

// ─────────────────────────────────────────────────────────────
// Signal Schema — what we extract from CE metadata + DRD
// ─────────────────────────────────────────────────────────────

export const CESignals = z.object({
  // Entry & logistics
  entryPoints: z.number().default(1),
  lanes: z.number().default(1),
  hasSecurityCheck: z.boolean().default(false),
  hasTicketDesk: z.boolean().default(false),
  
  // Crowd & capacity
  crowdVariability: z.enum(["none", "low", "medium", "high"]).default("medium"),
  fixedCapacity: z.boolean().default(false),
  unlimitedCapacity: z.boolean().default(false),
  typicalWaitMinutes: z.number().optional(),
  
  // Duration
  fixedDuration: z.boolean().default(false),
  typicalDurationMinutes: z.object({
    min: z.number(),
    max: z.number(),
  }).optional(),
  
  // Tickets
  ticketTiers: z.array(z.string()).default([]),
  hasSkipTheLine: z.boolean().default(false),
  hasGuidedTour: z.boolean().default(false),
  hasComboOption: z.boolean().default(false),
  priceGapPercent: z.number().default(0),
  
  // Spatial
  hasNamedZones: z.boolean().default(false),
  hasMultipleFloors: z.boolean().default(false),
  hasMultipleGalleries: z.boolean().default(false),
  crowdVariesByZone: z.boolean().default(false),
  
  // Transport
  transportModes: z.array(z.string()).default([]),
  
  // History
  historicalDepth: z.enum(["none", "shallow", "medium", "deep"]).default("medium"),
  keyTurningPoints: z.number().default(0),
  
  // Rules
  hasDressCode: z.boolean().default(false),
  hasProhibitedItems: z.boolean().default(false),
  hasPhotoRestrictions: z.boolean().default(false),
  
  // Weather/season
  weatherSensitivity: z.enum(["none", "low", "medium", "high"]).default("low"),
  hasSeasonalClosure: z.boolean().default(false),
  
  // Subcategory
  subcategory: z.string(),
  category: z.string(),
});

export type CESignals = z.infer<<typeof CESignals>;

// ─────────────────────────────────────────────────────────────
// DRD Signals — extracted from research document
// ─────────────────────────────────────────────────────────────

export const DRDSignals = z.object({
  // Queue data
  standardWaitRange: z.object({ min: z.number(), max: z.number() }).optional(),
  skipWaitRange: z.object({ min: z.number(), max: z.number() }).optional(),
  queueConfidence: z.number().min(0).max(1).optional(),
  
  // Sell-out data
  sellOutRisk: z.enum(["none", "low", "medium", "high"]).optional(),
  typicalBookingWindowDays: z.number().optional(),
  
  // Access data
  hasUnderground: z.boolean().optional(),
  requiresAddOn: z.boolean().optional(),
  hasArenaFloor: z.boolean().optional(),
  
  // Wildlife/sighting
  sightingProbability: z.record(z.number()).optional(), // month -> probability
  
  // Departure/light
  hasSunsetSlot: z.boolean().optional(),
  seasonalShift: z.boolean().optional(),
  hasMultipleDepartures: z.boolean().optional(),
  
  // Route
  hasNamedRoute: z.boolean().optional(),
  hasRouteVariants: z.boolean().optional(),
  hasMultipleSegments: z.boolean().optional(),
  
  // Seating/zones
  hasDeckZones: z.boolean().optional(),
  hasPriceTiers: z.boolean().optional(),
  hasSeatedVenue: z.boolean().optional(),
  
  // Gallery/room
  hasNamedGalleries: z.boolean().optional(),
  crowdVariesByGallery: z.boolean().optional(),
  
  // Product variants
  hasMultipleProducts: z.boolean().optional(),
  hasDayAndNight: z.boolean().optional(),
  hasHopOnOffVariant: z.boolean().optional(),
  hasDynamicPricing: z.boolean().optional(),
  
  // Activity
  hasHeadlineAttraction: z.boolean().optional(),
  hasMultipleSites: z.boolean().optional(),
  hasMultipleTrails: z.boolean().optional(),
  hasMultipleFormats: z.boolean().optional(),
  
  // Boarding/pressure
  hasBoardingPressureData: z.boolean().optional(),
  
  // Confidence
  overallConfidence: z.number().min(0).max(1),
});

export type DRDSignals = z.infer<<typeof DRDSignals>;

// ─────────────────────────────────────────────────────────────
// Question Definition
// ─────────────────────────────────────────────────────────────

export const QuestionDef = z.object({
  id: z.string(),
  
  // The visitor-facing question
  question: z.string(),
  
  // Chart type that answers it
  chartType: z.string(),
  
  // Which layer this belongs to
  layer: z.enum(["universal", "category", "ce_specific"]),
  
  // For universal: what blocks it
  blockWhen: z.record(z.any()).optional(),
  
  // For category: which subcategories
  applicableSubcategories: z.array(z.string()).optional(),
  
  // For category/ce: what triggers it
  trigger: z.object({
    ceSignals: z.record(z.any()).optional(),
    drdSignals: z.record(z.any()).optional(),
  }).optional(),
  
  // Category-specific overrides
  categoryOverrides: z.record(
    z.string(), // subcategory name
    z.object({
      question: z.string().optional(),
      chartType: z.string().optional(),
      priority: z.enum(["low", "medium", "high"]).optional(),
    })
  ).optional(),
  
  // Scoring function
  score: z.function().args(CESignals, DRDSignals).returns(z.number()).optional(),
  
  // Default score if no custom function
  defaultScore: z.number().min(0).max(100).default(50),
  
  // Page types this question appears on
  pageTypes: z.array(z.string()).default(["plan-your-visit"]),
  
  // Whether this is a "must have" for certain page types
  requiredFor: z.array(z.string()).default([]),
});

export type QuestionDef = z.infer<<typeof QuestionDef>;
