/**
 * Entrance Map — Chart Spec
 * 
 * An interactive map showing all entrances with:
 * - Location coordinates (relative or absolute)
 * - Wait time indicators
 * - Accessibility markers
 * - Landmark references
 * - "Best for" tags
 */

import { z } from "zod";

export const EntranceMapSpec = z.object({
  type: z.literal("entrance_map"),
  
  // Map configuration
  map: z.object({
    // If you have an actual map image/SVG
    baseMapUrl: z.string().optional(),
    // Or relative coordinate system (0-100)
    coordinateSystem: z.enum(["relative", "absolute"]).default("relative"),
    bounds: z.object({
      north: z.number(),
      south: z.number(),
      east: z.number(),
      west: z.number(),
    }).optional(), // For absolute (lat/lng)
  }),
  
  // Entrances
  entrances: z.array(z.object({
    id: z.string(),
    name: z.string(),              // "Main Entrance - Via Ricasoli"
    shortName: z.string(),         // "Main" — for map labels
    coordinates: z.object({
      x: z.number().min(0).max(100), // For relative
      y: z.number().min(0).max(100),
    }),
    
    // Current status
    status: z.enum(["open", "closed", "limited_access", "members_only"]),
    currentWaitMinutes: z.number().min(0).optional(),
    typicalWaitMinutes: z.number().min(0),
    
    // Accessibility
    accessibility: z.object({
      wheelchairAccessible: z.boolean(),
      hasRamp: z.boolean(),
      hasElevator: z.boolean(),
      stepFree: z.boolean(),
      accessibleBathroomNearby: z.boolean().optional(),
    }),
    
    // Best for whom
    bestFor: z.array(z.enum([
      "first_timers",
      "return_visitors",
      "families",
      "seniors",
      "mobility_limited",
      "large_groups",
      "solo_travelers",
      "rush_visitors",
    ])),
    
    // How to reach
    transit: z.array(z.object({
      mode: z.enum(["walk", "bus", "tram", "metro", "taxi", "bike"]),
      from: z.string(),            // "Santa Maria Novella Station"
      durationMinutes: z.number(),
      cost: z.object({
        amount: z.number(),
        currency: z.string().default("EUR"),
      }).optional(),
      instructions: z.string(),    // "Walk north on Via dei Fossi"
    })),
    
    // Security/bag check at this entrance
    securityLevel: z.enum(["none", "light", "standard", "strict"]),
    prohibitedItemsNote: z.string().optional(),
    
    // Visual styling
    color: z.string().optional(),   // Override default color
    icon: z.enum(["star", "clock", "wheelchair", "family", "group", "bolt"]).optional(),
    
    // Editorial
    tip: z.string().optional(),    // "Arrive 15 min early — security is strict here"
  })).min(1).max(6),
  
  // Landmarks for orientation
  landmarks: z.array(z.object({
    name: z.string(),
    coordinates: z.object({ x: z.number(), y: z.number() }),
    icon: z.enum(["landmark", "transit", "food", "restroom", "info"]),
  })).optional(),
  
  // Recommended entrance (highlighted)
  recommendedEntranceId: z.string().optional(),
  
  // Legend / key
  showLegend: z.boolean().default(true),
  
  // Interactivity
  interactive: z.object({
    clickable: z.boolean().default(true),
    showTooltips: z.boolean().default(true),
    zoomable: z.boolean().default(false),
  }).default({}),
  
  // Editorial overlay
  headline: z.string().default("Entrance Locations"),
  subheadline: z.string().default("Where to go and what to expect at each door"),
});
