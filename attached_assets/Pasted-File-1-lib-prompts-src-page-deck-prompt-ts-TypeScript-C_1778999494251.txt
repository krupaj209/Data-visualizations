/**
 * Page Deck Generation Prompt
 * 
 * The master prompt that guides Gemini to build a complete page deck.
 * This replaces the flat "select questions" prompt with a structured
 * narrative approach.
 */

export const PAGE_DECK_SYSTEM_PROMPT = `You are Viz Studio's Page Deck Architect.

Your job: Given a Combined Entity (CE) and a page type, design a complete visual narrative — a "deck" of sections that tells the right story for that page.

## Input Format

You will receive:
1. CE metadata (name, type, visitor volume, features)
2. Page type (plan-your-visit, skip-the-line, entrances, history, etc.)
3. Deep Research Document (DRD) — structured research about this CE
4. Existing charts already generated for this CE
5. Question Graph — available questions with their page-type variants

## Output Format

Return a JSON object matching this structure:

{
  "ceSlug": "string",
  "pageType": "string",
  "sections": [
    {
      "id": "string",           // e.g., "timing-when-to-go"
      "archetype": "string",    // timing, duration, logistics, value, experience, context, practical, social, navigation
      "name": "string",         // Human-readable section name
      "questionId": "string",   // Selected question from the graph
      "chartType": "string",    // Which chart type to generate
      "chartSpec": { ... },     // The actual chart specification (see below)
      "editorial": {
        "headline": "string",
        "subheadline": "string",
        "cta": { "text": "string", "variant": "primary|secondary|ghost" },
        "confidenceBadge": "high|medium|low",
        "tip": { "text": "string" },
        "warning": { "text": "string", "severity": "info|warning|alert" }
      },
      "layout": "full|half|third|hero",
      "aiRationale": "string"   // Why you chose this question and chart
    }
  ],
  "missingSections": [
    {
      "archetype": "string",
      "reason": "string",
      "fallbackMessage": "string"
    }
  ]
}

## Rules

### 1. Page-Type Narrative Arc
Each page type has a specific visitor journey. Follow it:

- **plan-your-visit**: timing → duration → logistics → experience → practical
  - Goal: Reduce anxiety, set expectations
  - Tone: Reassuring, practical, insider-smart
  - Must include: timing + duration sections

- **skip-the-line**: value → timing → practical → social-proof
  - Goal: Justify premium price with concrete proof
  - Tone: Confident, evidence-backed, urgency-tinged
  - Must include: at least one value section (time savings or ROI)

- **entrances**: navigation → timing → practical → value
  - Goal: Eliminate arrival confusion
  - Tone: Precise, map-oriented, stress-reducing
  - Must include: navigation section

- **history**: context → experience → social
  - Goal: Create emotional resonance
  - Tone: Narrative, reverent, surprising
  - Must include: context section

### 2. Question Selection Logic

For each section slot:
1. Check the section's \`questionPool\` — only these questions are eligible
2. Score each question:
   - Priority from \`pageVariants[pageType].priority\` (1 = must-have)
   - CE signal match (e.g., if CE hasMultipleEntrances, boost entrance_map)
   - DRD confidence for the archetype
   - Preferred chart type availability
3. Resolve conflicts: if two questions conflict, keep higher priority
4. Respect dependencies: if question A requires question B, ensure B is selected or already exists

### 3. Chart Spec Generation

Generate the actual chart specification based on the chart type. Each type has a strict schema:

- **weekly_pattern**: 7 days with crowd scores 0-100
- **hourly_heatmap**: 24 hours × 7 days matrix with crowd scores
- **duration_stat**: visit durations by interest level (express, standard, deep)
- **rules_checklist**: categorized rules with severity and status
- **entrance_map**: entrance locations with coordinates, wait times, accessibility
- **floor_plan_flow**: rooms with coordinates, crowd levels, must-see ratings, routes
- **transit_options**: ranked transport modes with segments
- **time_value_matrix**: visitor types × scenarios with ROI verdicts
- **queue_compare**: standard vs premium wait times
- **ticket_ladder**: ticket types with features and prices
- **story_timeline**: historical events with dates and significance

### 4. Editorial Generation

Every chart needs editorial overlay:

- **Headline**: Use the page-type variant from the question graph. Replace variables like {ceName}, {bestDay}, {bestTime}
- **Subheadline**: Explain data source and recency
- **CTA**: Contextual action. Primary for booking, secondary for exploration
- **Confidence badge**: "high" if DRD confidence > 0.8 and data is recent, "medium" if 0.5-0.8, "low" if < 0.5
- **Tip**: One insider insight that most visitors don't know
- **Warning**: Only if there's a genuine concern (strict security, seasonal closure, etc.)

### 5. Reuse Existing Charts

If the CE already has a chart that answers this question (even if the chart type differs):
- Reuse it by aliasing: e.g., if we need \`story_timeline\` but have \`history_timeline\`, use the existing chart
- Set \`chartSpec.status = "reused"\` and reference the existing chart ID
- Generate new editorial overlay for the new page context

### 6. Graceful Degradation

If DRD confidence is too low for a section:
- Add to \`missingSections\` with a helpful fallback message
- Never generate low-confidence charts — quality over completeness

## Example

Input: CE = "galleria-dellaccademia", pageType = "plan-your-visit"

Output sections:
1. timing-when-to-go → weekly_pattern (crowd by day)
2. duration-how-long → duration_stat (visit length by interest)
3. logistics-what-to-bring → rules_checklist (security, dress code)
4. experience-inside-flow → floor_plan_flow (room-by-room route)
5. practical-ticket-types → ticket_ladder (compare options)

Missing: logistics-getting-there (insufficient transit data in DRD)

Return ONLY the JSON. No markdown, no explanation.`;

export const PAGE_DECK_USER_PROMPT_TEMPLATE = `
## CE Metadata
Slug: {ceSlug}
Name: {ceName}
Category: {category}
Visitor Volume: {visitorVolume}
Features: {features}

## Page Type
{pageType}

## Deep Research Document
{drdContent}

## Existing Charts
{existingCharts}

## Question Graph
{questionGraph}

## Page Deck Template
{deckTemplate}

Generate the complete page deck JSON now.
`;

export function buildPageDeckPrompt(params: {
  ceSlug: string;
  ceName: string;
  category: string;
  visitorVolume: string;
  features: string[];
  pageType: string;
  drdContent: string;
  existingCharts: any[];
  questionGraph: any;
  deckTemplate: any;
}): { system: string; user: string } {
  const user = PAGE_DECK_USER_PROMPT_TEMPLATE
    .replace("{ceSlug}", params.ceSlug)
    .replace("{ceName}", params.ceName)
    .replace("{category}", params.category)
    .replace("{visitorVolume}", params.visitorVolume)
    .replace("{features}", params.features.join(", "))
    .replace("{pageType}", params.pageType)
    .replace("{drdContent}", params.drdContent.slice(0, 15000)) // Cap DRD length
    .replace("{existingCharts}", JSON.stringify(params.existingCharts, null, 2))
    .replace("{questionGraph}", JSON.stringify(params.questionGraph, null, 2))
    .replace("{deckTemplate}", JSON.stringify(params.deckTemplate, null, 2));

  return { system: PAGE_DECK_SYSTEM_PROMPT, user };
}

/**
 * Research Pipeline v2
 * 
 * Uses buildDeck() + new prompts for page-type-aware generation.
 * Runs alongside v1 pipeline, can be toggled per-CE.
 */

import { buildDeck } from "@workspace/page-decks";
import { buildPageDeckPrompt, getChartPrompt, EDITORIAL_OVERLAY_PROMPT } from "@workspace/prompts";
import { generateText } from "@workspace/integrations-gemini-ai";
import { db } from "@workspace/db";
import { charts, ces, drds } from "@workspace/db/schema";
import { eq } from "drizzle-orm";

interface PipelineV2Options {
  ceSlug: string;
  pageTypes: string[];
  useExistingCharts: boolean;
}

export async function runPipelineV2(options: PipelineV2Options) {
  const { ceSlug, pageTypes, useExistingCharts = true } = options;

  // Fetch CE
  const ce = await db.query.ces.findFirst({ where: eq(ces.slug, ceSlug) });
  if (!ce) throw new Error(`CE not found: ${ceSlug}`);

  // Fetch DRD
  const drd = await db.query.drds.findFirst({ where: eq(drds.ceSlug, ceSlug) });

  // Fetch existing charts
  const existingCharts = useExistingCharts 
    ? await db.query.charts.findMany({ where: eq(charts.ceSlug, ceSlug) })
    : [];

  const results = [];

  for (const pageType of pageTypes) {
    // Build deck structure
    const deck = await buildDeck(
      {
        slug: ce.slug,
        name: ce.name,
        category: ce.category || "museum",
        visitorVolume: (ce.visitorVolume as any) || "high",
        hasSeasonalVariation: true,
        hasMultipleEntrances: ce.hasMultipleEntrances || false,
        hasSkipTheLine: ce.hasSkipTheLine || true,
        hasAudioGuide: ce.hasAudioGuide || false,
        hasGuidedTours: ce.hasGuidedTours || false,
        hasRestrictedItems: ce.hasRestrictedItems || true,
        hasAccessibilityNeeds: ce.hasAccessibilityNeeds || false,
        typicalVisitDurationMin: ce.typicalVisitDurationMin || 60,
        typicalVisitDurationMax: ce.typicalVisitDurationMax || 180,
        subProductCount: ce.subProductCount || 1,
        drdConfidence: drd?.confidenceScores || {},
      },
      pageType as any,
      drd?.content || "",
      existingCharts
    );

    // For each pending section, trigger AI generation
    for (const section of deck.sections) {
      if (section.chartSpec.status === "pending-generation") {
        const chartPrompt = getChartPrompt(section.chartType);
        
        if (chartPrompt) {
          // Generate chart spec with Gemini
          const prompt = chartPrompt.replace("{ceName}", ce.name);
          
          const result = await generateText({
            prompt,
            system: "You are a data visualization expert. Generate ONLY valid JSON matching the specified schema. No markdown, no explanation.",
            temperature: 0.3,
          });

          // Parse and validate
          const chartSpec = JSON.parse(result.text);
          
          // Generate editorial overlay
          const editorialPrompt = EDITORIAL_OVERLAY_PROMPT
            .replace("{chartType}", section.chartType)
            .replace("{pageType}", pageType)
            .replace("{ceName}", ce.name)
            .replace("{chartData}", JSON.stringify(chartSpec))
            .replace("{ceData}", JSON.stringify(ce))
            .replace("{visitorContext}", "{}");

          const editorialResult = await generateText({
            prompt: editorialPrompt,
            system: "You are a travel content editor. Generate ONLY valid JSON. No markdown.",
            temperature: 0.5,
          });

          const editorial = JSON.parse(editorialResult.text);

          // Save to DB
          await db.insert(charts).values({
            ceSlug,
            questionId: section.question.questionId,
            chartType: section.chartType,
            spec: chartSpec,
            editorial,
            pageType,
            sectionId: section.id,
            status: "published",
            sortOrder: 0,
          });

          results.push({
            sectionId: section.id,
            chartType: section.chartType,
            status: "generated",
          });
        }
      }
    }
  }

  return results;
}
