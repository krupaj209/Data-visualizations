const MonthChart = () => {
  const [metric, setMetric] = useState("crowd");
  const metrics = {
    crowd: { label: "Crowd Level", color: palette.terracotta, desc: "How busy the gallery is (1=quiet, 10=packed)" },
    weather: { label: "Weather Score", color: palette.gold, desc: "Florence weather quality (1=poor, 10=ideal)" },
    price: { label: "Price Index", color: palette.green, desc: "Relative ticket demand & pricing pressure" },
  };
  const m = metrics[metric];

  const bestMonth = monthData.reduce((a, b) =>
    metric === "crowd" ? (a[metric] < b[metric] ? a : b) :
    (a[metric] > b[metric] ? a : b)
  );

  return (
    <Card>
      <SectionHeader number={1} title="When should I visit?" subtitle="Crowd levels, weather quality, and pricing pressure by month — all in one view." badge="Best Time" />
      <div style={{ display: "flex", gap: 8, marginBottom: 24, flexWrap: "wrap" }}>
        {Object.entries(metrics).map(([key, val]) => (
          <button key={key} onClick={() => setMetric(key)} style={{
            padding: "7px 16px", borderRadius: 20, border: "none", cursor: "pointer",
            fontFamily: "'DM Sans', sans-serif", fontWeight: 600, fontSize: 13,
            background: metric === key ? val.color : palette.border,
            color: metric === key ? "#fff" : palette.muted,
            transition: "all 0.2s"
          }}>{val.label}</button>
        ))}
      </div>
      <p style={{ fontSize: 13, color: palette.muted, marginBottom: 16 }}>{m.desc}</p>
      <ResponsiveContainer width="100%" height={200}>
        <BarChart data={monthData} barSize={28}>
          <XAxis dataKey="label" tick={{ fontSize: 12, fill: palette.muted }} axisLine={false} tickLine={false} />
          <YAxis hide domain={[0, 10]} />
          <Tooltip
            contentStyle={{ borderRadius: 8, border: `1px solid ${palette.border}`, fontSize: 13 }}
            cursor={{ fill: palette.highlight }}
          />
          <Bar dataKey={metric} radius={[4, 4, 0, 0]}>
            {monthData.map((entry, i) => (
              <Cell key={i} fill={entry.label === bestMonth.label ? m.color : m.color + "55"} />
            ))}
          </Bar>
        </BarChart>
      </ResponsiveContainer>
      <div style={{ marginTop: 16, display: "flex", gap: 12, flexWrap: "wrap" }}>
        {metric === "crowd"
          ? <InsightPill emoji="✅" text="Jan–Feb and Nov are the least crowded months. Avoid July–August." />
          : metric === "weather"
          ? <InsightPill emoji="🌤" text="May, September and October offer the best weather for visiting." />
          : <InsightPill emoji="💶" text="Jan–Mar sees lowest pricing pressure. Summer months see highest demand." />
        }
      </div>
    </Card>
  );
};

const LeadTimeChart = () => {
  const cumulative = leadTimeData.reduce((acc, item, i) => {
    const prev = i === 0 ? 0 : acc[i - 1].cum;
    acc.push({ ...item, cum: prev + item.pct });
    return acc;
  }, []);
  return (
    <Card>
      <SectionHeader number={5} title="How far ahead should I book?" subtitle="Distribution of when visitors actually purchase their tickets before the visit date." badge="Booking Lead" />
      <div style={{ display: "flex", gap: 6, marginBottom: 20, alignItems: "flex-end" }}>
        {leadTimeData.map((d, i) => (
          <div key={i} style={{ flex: 1, textAlign: "center" }}>
            <div style={{
              height: 100, display: "flex", alignItems: "flex-end", justifyContent: "center"
            }}>
              <div style={{
                width: "75%",
                height: `${d.pct * 4}px`,
                background: d.range === "15–30 days" ? palette.terracotta
                  : d.range === "Same day" ? palette.stone + "60"
                  : palette.terracotta + "55",
                borderRadius: "3px 3px 0 0", position: "relative"
              }}>
                {d.range === "15–30 days" && (
                  <div style={{
                    position: "absolute", top: -22, left: "50%", transform: "translateX(-50%)",
