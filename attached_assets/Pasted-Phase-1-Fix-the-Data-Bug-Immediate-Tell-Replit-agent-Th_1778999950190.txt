// artifacts/viz-studio/src/components/AttentionBanner.tsx

import React from "react";

interface AttentionItem {
  id: string;
  type: "drafts" | "stale" | "error" | "alert";
  message: string;
  ceSlug?: string;
  count?: number;
  actionLabel: string;
  actionUrl?: string;
}

interface Props {
  items: AttentionItem[];
  onDismiss: (id: string) => void;
  onAction: (item: AttentionItem) => void;
}

export const AttentionBanner: React.FC<<Props> = ({ items, onDismiss, onAction }) => {
  if (items.length === 0) return null;

  return (
    <div className="mb-6 bg-amber-50 border border-amber-200 rounded-xl p-4">
      <div className="flex items-center justify-between mb-3">
        <h3 className="text-sm font-semibold text-amber-900">
          ⚠️ Attention ({items.length})
        </h3>
        <button 
          onClick={() => items.forEach(i => onDismiss(i.id))}
          className="text-xs text-amber-600 hover:text-amber-800"
        >
          Dismiss all
        </button>
      </div>
      
      <div className="space-y-2">
        {items.map(item => (
          <div key={item.id} className="flex items-center justify-between bg-white rounded-lg p-3">
            <div className="flex items-center gap-3">
              <span className={
                item.type === "drafts" ? "text-blue-500" :
                item.type === "stale" ? "text-amber-500" :
                "text-red-500"
              }>
                {item.type === "drafts" && "📝"}
                {item.type === "stale" && "⏰"}
                {item.type === "error" && "❌"}
                {item.type === "alert" && "⚠️"}
              </span>
              <span className="text-sm text-gray-700">{item.message}</span>
            </div>
            <div className="flex items-center gap-2">
              <button
                onClick={() => onAction(item)}
                className="px-3 py-1.5 bg-purple-600 text-white rounded-lg text-xs font-medium hover:bg-purple-700"
              >
                {item.actionLabel}
              </button>
              <button
                onClick={() => onDismiss(item.id)}
                className="text-gray-400 hover:text-gray-600"
              >
                ✕
              </button>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
};

// artifacts/viz-studio/src/components/CECard.tsx

import React from "react";

interface CE {
  slug: string;
  name: string;
  city: string;
  country: string;
  chartCount: number;
  draftCount: number;
  status: "locked" | "published" | "stale" | "drafts_waiting";
  drdAgeDays: number;
  embedViewsChange?: number; // percentage, e.g. 12 for +12%
  thumbnailUrl?: string;
}

interface Props {
  ce: CE;
  onPreview: (slug: string) => void;
  onCopyEmbed: (slug: string) => void;
  onEdit: (slug: string) => void;
}

export const CECard: React.FC<<Props> = ({ ce, onPreview, onCopyEmbed, onEdit }) => {
  const statusConfig = {
    locked: { bg: "bg-purple-100", text: "text-purple-700", label: "LOCKED" },
    published: { bg: "bg-green-100", text: "text-green-700", label: "PUBLISHED" },
    stale: { bg: "bg-amber-100", text: "text-amber-700", label: "STALE" },
    drafts_waiting: { bg: "bg-blue-100", text: "text-blue-700", label: "DRAFTS" },
  };

  const status = statusConfig[ce.status];
  
  const drdColor = ce.drdAgeDays < 7 ? "text-green-600" : 
                   ce.drdAgeDays < 30 ? "text-amber-600" : 
                   "text-red-600";

  return (
    <div className="bg-white border border-gray-200 rounded-xl p-4 hover:shadow-md transition-shadow group">
      {/* Header */}
      <div className="flex items-start justify-between mb-3">
        <div className="flex items-center gap-3">
          {ce.thumbnailUrl ? (
            <img src={ce.thumbnailUrl} alt={ce.name} className="w-10 h-10 rounded-lg object-cover" />
          ) : (
            <div className="w-10 h-10 bg-gray-100 rounded-lg flex items-center justify-center text-lg">
              🏛️
            </div>
          )}
          <div>
            <h3 className="font-semibold text-gray-900">{ce.name}</h3>
            <p className="text-xs text-gray-500">{ce.city}, {ce.country}</p>
          </div>
        </div>
        <span className={`px-2 py-0.5 rounded-full text-xs font-medium ${status.bg} ${status.text}`}>
          {status.label}
        </span>
      </div>

      {/* Stats */}
      <div className="flex items-center gap-4 mb-3 text-sm">
        <div className="flex items-center gap-1">
          <span className="w-2 h-2 rounded-full bg-green-500" />
          <span className="text-gray-700">{ce.chartCount} charts</span>
          {ce.draftCount > 0 && (
            <span className="text-blue-600">· {ce.draftCount} drafts</span>
          )}
        </div>
        
        {ce.embedViewsChange !== undefined && (
          <div className={`flex items-center gap-1 ${
            ce.embedViewsChange > 0 ? "text-green-600" : 
            ce.embedViewsChange < 0 ? "text-red-600" : 
            "text-gray-500"
          }`}>
            <span>{ce.embedViewsChange > 0 ? "↑" : ce.embedViewsChange < 0 ? "↓" : "→"}</span>
            <span>{Math.abs(ce.embedViewsChange)}% views</span>
          </div>
        )}
      </div>

      {/* DRD freshness */}
      <div className={`text-xs mb-3 ${drdColor}`}>
        ◐ DRD {ce.drdAgeDays}d old
      </div>

      {/* Actions */}
      <div className="flex items-center gap-2 opacity-0 group-hover:opacity-100 transition-opacity">
        <button
          onClick={() => onPreview(ce.slug)}
          className="px-3 py-1.5 bg-gray-100 text-gray-700 rounded-lg text-xs font-medium hover:bg-gray-200"
        >
          Preview
        </button>
        <button
          onClick={() => onCopyEmbed(ce.slug)}
          className="px-3 py-1.5 bg-gray-100 text-gray-700 rounded-lg text-xs font-medium hover:bg-gray-200"
        >
          Copy embed
        </button>
        <button
          onClick={() => onEdit(ce.slug)}
          className="px-3 py-1.5 bg-purple-50 text-purple-700 rounded-lg text-xs font-medium hover:bg-purple-100"
        >
          Edit charts
        </button>
      </div>
    </div>
  );
};

// artifacts/viz-studio/src/components/CityClusters.tsx

import React from "react";

interface CityCluster {
  city: string;
  country: string;
  ceCount: number;
  publishedCount: number;
  draftCount: number;
  color: string;
}

interface Props {
  clusters: CityCluster[];
  onSelect: (city: string) => void;
}

export const CityClusters: React.FC<<Props> = ({ clusters, onSelect }) => {
  return (
    <div className="mt-6">
      <h3 className="text-sm font-semibold text-gray-700 mb-3">City clusters</h3>
      <div className="flex flex-wrap gap-2">
        {clusters.map(cluster => (
          <button
            key={cluster.city}
            onClick={() => onSelect(cluster.city)}
            className="flex items-center gap-2 px-3 py-2 bg-white border border-gray-200 rounded-xl hover:border-purple-300 hover:shadow-sm transition-all"
          >
            <span 
              className="w-2 h-2 rounded-full" 
              style={{ backgroundColor: cluster.color }}
            />
            <div className="text-left">
              <div className="text-sm font-medium text-gray-900">{cluster.city}</div>
              <div className="text-xs text-gray-500">
                {cluster.ceCount} CEs · {cluster.publishedCount} published
                {cluster.draftCount > 0 && ` · ${cluster.draftCount} drafts`}
              </div>
            </div>
          </button>
        ))}
      </div>
    </div>
  );
};
